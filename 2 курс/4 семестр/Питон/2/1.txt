def f(s, c):
    return s.lower().count(c)


def count_characters(string):
    new_string = string.lower().replace(' ', '')
    dictionary = {}
    for char in new_string:
        if char in dictionary:
            dictionary[char] += 1
        else:
            dictionary[char] = 1
    return dictionary
